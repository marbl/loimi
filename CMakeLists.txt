cmake_minimum_required(VERSION 3.18)
project(hpc_consensus VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(HPC_ENABLE_SANITIZERS "Instrument project code with address/undefined sanitizers" OFF)
set(VERKKO_ROOT "" CACHE PATH "Verkko checkout with build/lib/libverkko.a")
# cnspack serialization is native-ABI dependent: compile against the same
# headers/build used for the packages rather than vendoring a guessed reader.
set(VUTIL "${VERKKO_ROOT}/src/canu/src/utility/src")
if(NOT EXISTS "${VERKKO_ROOT}/build/lib/libverkko.a")
  message(FATAL_ERROR "Set VERKKO_ROOT to a built Verkko checkout")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(SYSTEM_DEPS REQUIRED IMPORTED_TARGET openssl libcurl liblzma)
find_package(ZLIB REQUIRED)
find_package(BZip2 REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED COMPONENTS CXX)
# The imported OpenMP target supplies compiler flags as well as the runtime;
# locating a libomp/libgomp filename alone is not portable across Linux/macOS.
add_library(consensus src/core.cpp src/models.cpp src/polish.cpp src/ends.cpp src/offsets.cpp
  src/experiments/borrowing.cpp)
target_include_directories(consensus PUBLIC src)
# Quote-only lookup avoids utility/math.H shadowing the system math.h on macOS.
target_compile_options(consensus PUBLIC "-iquote${VUTIL}" "-iquote${VERKKO_ROOT}/src/canu/src/stores")
# zlib/bzip2 are link-only dependencies of the native archive. Their macOS
# imported targets can inject SDK/usr/include ahead of libc++ wrappers when
# using Homebrew Clang, so do not propagate their unused header directories.
target_link_libraries(consensus PUBLIC "${VERKKO_ROOT}/build/lib/libverkko.a" PkgConfig::SYSTEM_DEPS ${ZLIB_LIBRARIES} ${BZIP2_LIBRARIES} Threads::Threads OpenMP::OpenMP_CXX)
if(HPC_ENABLE_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "Sanitizers require Clang or GCC")
  endif()
  # Propagate to project executables; the prebuilt Verkko archive is not instrumented.
  target_compile_options(consensus PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(consensus PUBLIC -fsanitize=address,undefined)
endif()
target_compile_options(consensus PRIVATE -Wall -Wextra -Wpedantic)
add_library(consensus-io src/io.cpp src/options.cpp)
target_link_libraries(consensus-io PUBLIC consensus)
target_compile_options(consensus-io PRIVATE -Wall -Wextra -Wpedantic)
add_executable(hpc-consensus src/main.cpp)
target_compile_options(hpc-consensus PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(hpc-consensus PRIVATE consensus-io)
install(TARGETS hpc-consensus RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
# Core tests need no input datasets. Integration generates small native packages
# with Verkko tools, and exercises CRAM only when samtools is available on PATH.
add_executable(core-tests tests/core.cpp)
target_link_libraries(core-tests PRIVATE consensus)
add_executable(extract-regions tests/extract_regions.cpp)
target_link_libraries(extract-regions PRIVATE consensus)
add_executable(apply-run-model tests/apply_run_model.cpp)
target_link_libraries(apply-run-model PRIVATE consensus)
# Diagnostic links the actual external Verkko graph implementation, retaining
# its upstream license and keeping it out of the production executable.
option(HPC_BUILD_GRAPH_PROBE "Build optional same-read Verkko graph diagnostic (needs Boost headers)" OFF)
if(HPC_BUILD_GRAPH_PROBE)
find_path(HPC_BOOST_INCLUDE_DIR boost/graph/adjacency_list.hpp HINTS /opt/homebrew/include /usr/local/include REQUIRED)
add_executable(hp-graph-probe tests/hp_graph_probe.cpp "${VERKKO_ROOT}/src/canu/src/utgcns/libpbutgcns/AlnGraphBoost.C")
target_include_directories(hp-graph-probe PRIVATE "${VERKKO_ROOT}/src/canu/src/utgcns/libpbutgcns")
target_include_directories(hp-graph-probe SYSTEM PRIVATE "${HPC_BOOST_INCLUDE_DIR}")
target_link_libraries(hp-graph-probe PRIVATE consensus)
# Verkko's bundled Boost predates C++17's removal of std::unary_function.
# The probe's ABI-compatible interface needs only C++14; production stays C++20.
set_target_properties(hp-graph-probe PROPERTIES CXX_STANDARD 14)
endif()
add_test(NAME core COMMAND core-tests)
add_executable(maintenance-tests tests/maintenance.cpp)
target_link_libraries(maintenance-tests PRIVATE consensus-io)
add_test(NAME maintenance COMMAND maintenance-tests)
add_executable(linked-event-tests tests/linked_events.cpp)
target_link_libraries(linked-event-tests PRIVATE consensus)
add_test(NAME linked-events COMMAND linked-event-tests)
add_executable(additive-tests tests/additive.cpp)
target_link_libraries(additive-tests PRIVATE consensus)
add_test(NAME additive COMMAND additive-tests)
add_executable(borrow-tests tests/borrow.cpp)
add_executable(hp-anchor-tests tests/hp_anchors.cpp)
target_link_libraries(hp-anchor-tests PRIVATE consensus)
add_test(NAME hp-anchors COMMAND hp-anchor-tests)
target_link_libraries(borrow-tests PRIVATE consensus)
add_test(NAME borrow COMMAND borrow-tests)
add_executable(apply-borrow tests/apply_borrow.cpp)
target_link_libraries(apply-borrow PRIVATE consensus)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME native-offset COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/offset_native.py" "$<TARGET_FILE:hpc-consensus>" "${VERKKO_ROOT}")
add_test(NAME native-anchors COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/anchor_native.py" "$<TARGET_FILE:hpc-consensus>" "${VERKKO_ROOT}")
add_test(NAME experiment-classification COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/experiments/test_classification.py")
add_test(NAME native-borrow COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/borrow_native.py" "$<TARGET_FILE:hpc-consensus>" "${VERKKO_ROOT}")
if(HPC_BUILD_GRAPH_PROBE)
add_test(NAME graph-probe COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/graph_probe.py" "$<TARGET_FILE:hp-graph-probe>")
endif()
add_test(NAME native-package COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:hpc-consensus>" "${VERKKO_ROOT}")
add_test(NAME calibration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/calibration.py")
endif()
